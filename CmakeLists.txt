cmake_minimum_required(VERSION 3.8)

set(PROJECT_NAME tecslog-demo)
project(${PROJECT_NAME} VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

set(PROG_LIBRARY "tecslog")

add_subdirectory(tecslog)
add_subdirectory(demo-app)
add_subdirectory(tests)

add_custom_target(destroy
  COMMAND ${CMAKE_COMMAND} -E remove_directory ${CMAKE_BINARY_DIR}
  COMMENT "Removing cmake build directory..."
)

enable_testing()

add_custom_target(test-lib
  COMMAND ${CMAKE_CTEST_COMMAND} -R test-lib --verbose
  DEPENDS test_lib
  WORKING_DIRECTORY ${CMAKE_BINARY_DIR}
  COMMENT "Running library unit tests..."
)

add_custom_target(test-demo
  COMMAND ${CMAKE_CTEST_COMMAND} -R test-demo --verbose
  DEPENDS test_demo
  WORKING_DIRECTORY ${CMAKE_BINARY_DIR}
  COMMENT "Running demo unit tests..."
)
