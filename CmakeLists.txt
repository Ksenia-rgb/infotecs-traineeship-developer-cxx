cmake_minimum_required(VERSION 3.8)

set(PROJECT_NAME tecslog-demo)
project(${PROJECT_NAME} VERSION 1.0 LANGUAGES CXX)

enable_testing()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

set(PROG_LIBRARY "tecslog")

add_subdirectory(tecslog)
add_subdirectory(demo-app)
add_subdirectory(tests)

add_custom_target(destroy
  COMMAND ${CMAKE_COMMAND} -E remove_directory ${CMAKE_BINARY_DIR}
  COMMENT "Removing cmake build directory..."
)

add_custom_target(tests
  DEPENDS test_tecslog_init test_tecslog_level test_tecslog_log
  COMMENT "Building all tests..."
)

add_custom_target(test-verbose
  COMMAND ${CMAKE_CTEST_COMMAND} --verbose
  DEPENDS test_tecslog_init test_tecslog_level test_tecslog_log
  WORKING_DIRECTORY ${CMAKE_BINARY_DIR}
  COMMENT "Running all tests with verbose output..."
)
